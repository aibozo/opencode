# PR 4 — MCP server: Context Pack + Span Reads + Skeleton + Search

## Objectives

* Implement a **real MCP server** over stdio (JSON-RPC 2.0).
* Expose the tools your agent will rely on:

  * `repo.graphBuild`
  * `repo.workingSet.select`
  * `repo.code.readSpan`
  * `repo.code.skeleton`
  * `repo.code.search`
  * `repo.info.stats` (handy debug)
* Enforce **safety**: root-relative paths only, deny secrets, clamp span sizes.
* Add **integration tests** that perform a mini hand-shake and call tools.
* Keep OpenAI-first config unchanged; OpenCode will discover the tools.

  task: z.string().min(1),

### `packages/repo-mcp/src/paths.ts` (path guards)

```ts
import { resolve, relative } from "path";
import { readFileSync } from "fs";

export function normalizeAndCheck(root: string, p: string) {
  const abs = resolve(root, p);
  const rel = relative(root, abs).replace(/\\/g, "/");
  if (rel.startsWith("..")) throw new Error(`Path escapes root: ${p}`);
  if (/(^|\/)\.(env|env\.local|aws|ssh|git)($|\/)/i.test(rel)) {
    throw new Error(`Access denied to sensitive path: ${rel}`);
  }
  return { abs, rel };
}

export function readSpan(abs: string, start: number, end: number, maxLines = 8000) {
  if (end < start) throw new Error("end < start");
  const lines = readFileSync(abs, "utf8").split(/\r?\n/);
  const a = Math.max(1, start) - 1;
  const b = Math.min(end, lines.length) - 1;
  const count = b - a + 1;
  if (count > maxLines) throw new Error(`span too large: ${count} > ${maxLines}`);
  return { text: lines.slice(a, b + 1).join("\n"), total: lines.length };
}
```

### `packages/repo-mcp/src/server.ts` (real MCP over stdio)

```ts
import { createStdioServer } from "@modelcontextprotocol/sdk/server/stdio";
import { Tool } from "@modelcontextprotocol/sdk/types";
import { z } from "zod";
import { ZSelectArgs, ZReadSpanArgs, ZSkeletonArgs, ZSearchArgs } from "./schemas";
import { buildIndex, ensureWatcher, getIndex } from "@opencode/repo-index/src/index";
import { selectWorkingSet } from "@opencode/repo-index/src/working-set";
import { skeletonizeFile } from "@opencode/repo-index/src/skeleton";
import { grepKeywords, keywordsFromTask } from "@opencode/repo-index/src/search";
import { normalizeAndCheck, readSpan } from "./paths";
import { resolve } from "path";
import crypto from "crypto";

const PKG = { name: "repo-mcp", version: "0.0.4" };

export async function startServer() {
  const root = process.env.REPO_ROOT ? resolve(process.env.REPO_ROOT) : process.cwd();

  const tools: Tool[] = [
    {
      name: "repo.graphBuild",
      description: "Build or refresh the repository index and start a file watcher.",
      inputSchema: { type: "object", properties: {}, additionalProperties: false }
    },
    {
      name: "repo.info.stats",
      description: "Return index summary counts for debugging.",
      inputSchema: { type: "object", properties: {}, additionalProperties: false }
    },
    {
      name: "repo.workingSet.select",
      description: "Select a token-budgeted Context Pack for a task.",
      inputSchema: ZSelectArgs.toJSON() as any
    },
    {
      name: "repo.code.readSpan",
      description: "Read a file slice by line numbers (inclusive).",
      inputSchema: ZReadSpanArgs.toJSON() as any
    },
    {
      name: "repo.code.skeleton",
      description: "Return a skeleton view and elision metadata for a file.",
      inputSchema: ZSkeletonArgs.toJSON() as any
    },
    {
      name: "repo.code.search",
      description: "Find lines matching keywords/patterns (simple grep).",
      inputSchema: ZSearchArgs.toJSON() as any
    }
  ];

  const server = createStdioServer(
    {
      name: PKG.name,
      version: PKG.version,
      tools
    },
    {
      // Tool router
      async callTool(name, args) {
        try {
          switch (name) {
            case "repo.graphBuild": {
              await buildIndex(root);
              ensureWatcher(root);
              const idx = getIndex();
              return { ok: true, root, updatedAt: idx.updatedAt, files: idx.files.length, symbols: idx.symbols.length, edges: idx.edges.length };
            }
            case "repo.info.stats": {
              const idx = getIndex();
              return { ok: true, root, files: idx.files.length, symbols: idx.symbols.length, edges: idx.edges.length, updatedAt: idx.updatedAt };
            }
            case "repo.workingSet.select": {
              const p = ZSelectArgs.parse(args ?? {});
              const { pack, debug } = await selectWorkingSet(root, p);
              return { pack, debug };
            }
            case "repo.code.readSpan": {
              const p = ZReadSpanArgs.parse(args ?? {});
              const { abs, rel } = normalizeAndCheck(root, p.path);
              const result = readSpan(abs, p.start, p.end);
              const sha1 = crypto.createHash("sha1").update(result.text).digest("hex");
              return { path: rel, start: p.start, end: p.end, lines: result.total, sha1, text: result.text };
            }
            case "repo.code.skeleton": {
              const p = ZSkeletonArgs.parse(args ?? {});
              const { abs, rel } = normalizeAndCheck(root, p.path);
              const sk = skeletonizeFile(root, abs);
              return { path: rel, text: sk.text, elisions: sk.elisions };
            }
            case "repo.code.search": {
              const p = ZSearchArgs.parse(args ?? {});
              const idx = getIndex();
              const files = (p.globs && p.globs.length)
                ? idx.files.filter(f => require("minimatch").minimatch(f, p.globs[0])) // simple single-glob support to start
                : idx.files;
              const ks = keywordsFromTask(p.pattern);
              const hits = grepKeywords(root, files, ks, 10).slice(0, p.maxHits ?? 100);
              return { hits: hits.map(h => ({ path: h.path.replace(root + "/", ""), line: h.line, text: h.text })) };
            }
          }
          throw new Error(`Unknown tool: ${name}`);
        } catch (e: any) {
          return { error: String(e?.message ?? e) };
        }
      }
    }
  );

  await server.connect();
}
```

## Tests (integration over MCP)

### `tests/mcp-tools-list.test.ts`

```ts
import { describe, it, expect } from "vitest";
import { spawn } from "child_process";

async function rpc(proc: any, msg: any) {
  proc.stdin.write(JSON.stringify(msg) + "\n");
  return await new Promise<string>(res => proc.stdout.once("data", (b: Buffer) => res(b.toString("utf8").trim())));
}

describe("MCP tools list", () => {
  it("exposes our repo tools", async () => {
    const proc = spawn("node", ["packages/repo-mcp/dist/bin/repo-mcp.js"], { stdio: ["pipe","pipe","pipe"] });

    // minimal handshake for stdio server (sdk prints a ready line or responds to 'tools/list')
    const list = await rpc(proc, { jsonrpc: "2.0", id: 1, method: "tools/list" });
    const resp = JSON.parse(list);
    expect(resp.id).toBe(1);
    const names = resp.result.tools.map((t: any) => t.name);
    for (const t of ["repo.graphBuild","repo.workingSet.select","repo.code.readSpan","repo.code.skeleton","repo.code.search","repo.info.stats"]) {
      expect(names).toContain(t);
    }
    proc.kill();
  }, 10000);
});
```

import { describe, it, expect } from "vitest";
import { spawn } from "child_process";
import { resolve } from "path";
import { readFileSync } from "fs";

async function rpc(proc: any, id: number, method: string, params?: any) {
  proc.stdin.write(JSON.stringify({ jsonrpc: "2.0", id, method, params }) + "\n");
  const line: string = await new Promise(res => proc.stdout.once("data", (b: Buffer) => res(b.toString("utf8").trim())));
  return JSON.parse(line);
}

describe("MCP workingSet.select", () => {
  it("returns a pack with full+skeletal under budget", async () => {
    const proc = spawn("node", ["packages/repo-mcp/dist/bin/repo-mcp.js"], { stdio: ["pipe","pipe","pipe"], env: { ...process.env, REPO_ROOT: resolve("fixtures/ts-mini") } });
    await rpc(proc, 1, "tools/call", { name: "repo.graphBuild", arguments: {} });
    const issue = readFileSync(resolve("fixtures/ts-mini/ISSUE.md"), "utf8");
    const r = await rpc(proc, 2, "tools/call", { name: "repo.workingSet.select", arguments: { task: issue, budgetTokens: 3000, radius: 1 } });
    expect(r.result.pack.full.length).toBeGreaterThan(0);
    expect(r.result.pack.budget.used).toBeLessThanOrEqual(r.result.pack.budget.target);
    proc.kill();
  }, 15000);
});

# Agent: semantic-compress — Tool usage (PR 4)

On each new task:
1) Call `repo.graphBuild` (idempotent; cheap).
2) Call `repo.workingSet.select` with the task text, `budgetTokens` (e.g., 64k), and `radius` (1..2).
3) Include in your prompt:
   - FULL files/ranges
   - SKELETON neighbors
   - NOTES (verbatim extract)
4) When you need more detail from a file:
   - Prefer `repo.code.readSpan(path, start, end)` for the smallest required lines.
   - Use `repo.code.skeleton(path)` to recall interfaces / signatures.
5) If a symbol or type is missing, repeat step (2) with a larger radius or targeted keywords.
