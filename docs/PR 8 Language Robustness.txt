# PR 8 — Language Robustness (+ Go)

## Objectives

* Add **Go** parsing + skeletonization using Tree-sitter.
* Teach the indexer to **incrementally** re-parse only changed files.
* Make imports **more resolvable** (TS path aliases, Python relative, Go module path → repo path).
* Add **graceful fallbacks** when parsing fails (we still produce a skeleton stub rather than crash).
* Extend working-set + watcher to cover `*.go`.

---

## Changes at a glance

* New parser: `tree-sitter-go`
* New files: `lang/go.ts`, `skeleton/go.ts`
* Update globs to include `**/*.go`
* Incremental index cache (per-file)
* Import resolution helpers:

  * **TS/JS**: read `tsconfig.json` (`baseUrl`, `paths`) for local path resolution
  * **Python**: resolve `from .foo` via filesystem; module names via package dirs
  * **Go**: read `go.mod` module name; map `import "<module>/<subpath>"` to repo path
* Working-set neighbor expansion considers new resolved **file→file** import edges
* Fixtures + tests for Go; tests for incremental rebuild correctness + fallbacks

## Indexer: Go support + incremental cache

### `packages/repo-index/src/lang/go.ts`

```ts
import Parser from "node-tree-sitter";
import Go from "tree-sitter-go";
import { readFileSync } from "fs";
import { relative } from "path";
import type { SymbolInfo, Edge } from "../types";

const parser = new Parser();
parser.setLanguage(Go as any);

export function parseGo(root: string, file: string): { symbols: SymbolInfo[]; edges: Edge[] } {
  const src = readFileSync(file, "utf8");
  const tree = parser.parse(src);

  const symbols: SymbolInfo[] = [];
  const edges: Edge[] = [];
  const rel = relative(root, file).replace(/\\/g, "/");
  const moduleName = rel.replace(/\.go$/, "").replace(/\//g, ".");

  const L = (n: any) => n.startPosition.row + 1;

  function scoped(name: string) {
    return `${moduleName}:${name}`;
  }

  // package decl
  symbols.push({
    file, lang: "go", name: `${moduleName}:package`, shortName: "package",
    kind: "module", startLine: 1, endLine: tree.rootNode.endPosition.row + 1
  });

  tree.rootNode.namedChildren?.forEach((n: any) => {
    if (n.type === "import_declaration") {
      // each import_spec has a string literal path
      n.namedChildren?.forEach((im: any) => {
        if (im.type === "import_spec") edges.push({ from: `${moduleName}:package`, to: im.text, type: "imports" });
      });
    }
    if (n.type === "function_declaration") {
      const name = n.childForFieldName("name")?.text ?? "anonymous";
      const recv = n.childForFieldName("receiver"); // method receiver => method
      const kind = recv ? "method" : "func";
      symbols.push({
        file, lang: "go", name: scoped(name), shortName: name, kind: kind as any,
        startLine: L(n), endLine: n.endPosition.row + 1
      });
      // naive calls: collect identifiers inside body
      n.descendantsOfType?.(["identifier"])?.forEach((id: any) => {
        edges.push({ from: scoped(name), to: id.text, type: "calls" });
      });
    }
    if (n.type === "type_declaration" || n.type === "const_declaration" || n.type === "var_declaration") {
      // summary symbol for types/consts/vars
      const nameNode = n.namedChildren?.find((c: any) => c.type === "type_spec" || c.type === "const_spec" || c.type === "var_spec")
        ?.childForFieldName("name");
      if (nameNode) {
        const nm = nameNode.text;
        symbols.push({
          file, lang: "go", name: scoped(nm), shortName: nm, kind: n.type.startsWith("type") ? "type" : n.type.startsWith("const") ? "const" : "var",
          startLine: L(n), endLine: n.endPosition.row + 1
        });
      }
    }
  });

  return { symbols, edges };
}
```

### `packages/repo-index/src/skeleton/go.ts`

```ts
import Parser from "node-tree-sitter";
import Go from "tree-sitter-go";
import { relative } from "path";
import { Skeleton, Elision } from "@opencode/shared/src/types";
import { sha1 } from "../skeleton";
import { readFileSync } from "fs";

const parser = new Parser();
parser.setLanguage(Go as any);

export function skeletonizeGo(root: string, file: string, src?: string): Skeleton {
  const text = src ?? readFileSync(file, "utf8");
  const tree = parser.parse(text);
  const lines = text.split(/\r?\n/);
  const rel = relative(root, file).replace(/\\/g, "/");

  const out: string[] = [];
  const elisions: Elision[] = [];
  const L = (n: any) => n.startPosition.row + 1;
  const slice = (a: number, b: number) => lines.slice(a, b + 1).join("\n");

  // package + imports verbatim
  tree.rootNode.namedChildren?.forEach((n: any) => {
    if (n.type === "package_clause" || n.type === "import_declaration") {
      out.push(slice(n.startPosition.row, n.endPosition.row));
    }
  });
  if (out.length) out.push("");

  function elide(kind: Elision["kind"], name: string | undefined, start: number, end: number) {
    const txt = slice(start, end);
    const e = { kind, name, startLine: start + 1, endLine: end + 1, linesElided: end - start + 1, sha1: sha1(txt) };
    elisions.push(e);
    return e;
  }

  tree.rootNode.namedChildren?.forEach((n: any) => {
    if (n.type === "function_declaration") {
      // header up to '{'
      const hdrRow = n.startPosition.row;
      const bodyStart = n.namedChildren?.find((c: any) => c.type === "block")?.startPosition.row ?? n.startPosition.row;
      out.push(lines[hdrRow]);
      out.push("{");
      const e = elide("func", n.childForFieldName("name")?.text ?? "fn", bodyStart + 1, n.endPosition.row - 1);
      out.push(`  /* ⟪ELIDED L${e.startLine}-${e.endLine}; ${e.linesElided} lines; sha1=${e.sha1}⟫ */`);
      out.push("}\n");
    }
    if (n.type === "type_declaration" || n.type === "const_declaration" || n.type === "var_declaration") {
      out.push(slice(n.startPosition.row, n.endPosition.row) + "\n");
    }
  });

  const textOut = [`// File: ${rel}`, ...out].join("\n").replace(/\n{3,}/g, "\n\n");
  return { path: rel, lang: "go", text: textOut, elisions };
}
```

### `packages/repo-index/src/index.ts` (add go, incremental cache)

```ts
import { statSync, readFileSync } from "fs";
import { parseGo } from "./lang/go";
// ...

type FileCache = {
  mtimeMs: number;
  size: number;
  hash: string;
  symbols: SymbolInfo[];
  edges: Edge[];
};
const CACHE = new Map<string, FileCache>();

function fileHash(absPath: string) {
  const c = readFileSync(absPath, "utf8");
  // tiny hash to detect changes quickly without bringing in a new dep
  let h = 0; for (let i=0;i<c.length;i++) { h = (h * 131 + c.charCodeAt(i)) >>> 0; }
  return String(h);
}

function parseOne(root: string, file: string, lang: Lang | undefined) {
  if (!lang) return { symbols: [], edges: [] };
  if (lang === "python") return parsePython(root, file);
  if (lang === "typescript") return parseTypeScript(root, file);
  if (lang === "javascript") return parseJavaScript(root, file);
  if (lang === "go") return parseGo(root, file);
  return { symbols: [], edges: [] };
}

function langOf(path: string): Lang | undefined {
  if (/\.(py)$/.test(path)) return "python";
  if (/\.(ts|tsx)$/.test(path)) return "typescript";
  if (/\.(js|jsx|mjs|cjs)$/.test(path)) return "javascript";
  if (/\.go$/.test(path)) return "go";
  return undefined;
}

export async function buildIndex(root: string, opts: BuildOptions = {}): Promise<RepoIndex> {
  const include = opts.include ?? ["**/*.{py,ts,tsx,js,jsx,go}"];
  const exclude = opts.exclude ?? ["**/node_modules/**","**/dist/**","**/build/**","**/.git/**"];
  const files = await fg(include, { cwd: root, ignore: exclude, absolute: true });

  const symbols: SymbolInfo[] = [];
  const edges: Edge[] = [];
  for (const file of files) {
    const lang = langOf(file);
    const st = statSync(file);
    const hash = fileHash(file);
    const prev = CACHE.get(file);
    if (prev && prev.mtimeMs === st.mtimeMs && prev.size === st.size && prev.hash === hash) {
      symbols.push(...prev.symbols); edges.push(...prev.edges); continue;
    }
    const parsed = parseOne(root, file, lang);
    CACHE.set(file, { mtimeMs: st.mtimeMs, size: st.size, hash, symbols: parsed.symbols, edges: parsed.edges });
    symbols.push(...parsed.symbols); edges.push(...parsed.edges);
  }
  CURRENT_INDEX = { root: resolve(root), files, symbols, edges, updatedAt: Date.now() };
  return CURRENT_INDEX;
}

export async function updateIndex(changedPaths: string[]): Promise<RepoIndex> {
  if (!CURRENT_INDEX.root) return CURRENT_INDEX;
  // Only reparse changed files (exists check omitted for brevity)
  const keepFiles = new Set(CURRENT_INDEX.files);
  const symbols: SymbolInfo[] = [];
  const edges: Edge[] = [];
  for (const f of keepFiles) {
    if (changedPaths.length && !changedPaths.includes(f)) {
      const prev = CACHE.get(f);
      if (prev) { symbols.push(...prev.symbols); edges.push(...prev.edges); continue; }
    }
    const lang = langOf(f);
    const st = statSync(f);
    const hash = fileHash(f);
    const parsed = parseOne(CURRENT_INDEX.root, f, lang);
    CACHE.set(f, { mtimeMs: st.mtimeMs, size: st.size, hash, symbols: parsed.symbols, edges: parsed.edges });
    symbols.push(...parsed.symbols); edges.push(...parsed.edges);
  }
  CURRENT_INDEX = { ...CURRENT_INDEX, symbols, edges, updatedAt: Date.now() };
  return CURRENT_INDEX;
}
```

### `packages/repo-index/src/watcher.ts` (include .go & pass changed path)

```ts
export function startWatcher(root: string, onChange: (paths: string[]) => Promise<void>) {
  const watcher = chokidar.watch(["**/*.{py,ts,tsx,js,jsx,go}"], {
    cwd: root,
    ignored: ["**/node_modules/**","**/dist/**","**/build/**","**/.git/**"],
    ignoreInitial: true
  });
  const queue = new Set<string>();
  const flush = async () => {
    if (!queue.size) return;
    const paths = Array.from(queue).map(p => resolve(root, p));
    queue.clear();
    await onChange(paths);
  };
  watcher.on("add", p => { queue.add(p); flush(); });
  watcher.on("change", p => { queue.add(p); flush(); });
  watcher.on("unlink", p => { queue.add(p); flush(); });
  return watcher;
}
```

## Import resolution (quality bump)

### `packages/repo-index/src/resolve.ts` (new)

```ts
import { existsSync, readFileSync } from "fs";
import { resolve, join, dirname } from "path";

export interface TSConfig { baseUrl?: string; paths?: Record<string, string[]>; }
export function loadTSConfig(root: string): TSConfig {
  try {
    const p = resolve(root, "tsconfig.json");
    if (!existsSync(p)) return {};
    const j = JSON.parse(readFileSync(p, "utf8"));
    return j.compilerOptions ?? {};
  } catch { return {}; }
}

export function resolveTSImport(root: string, fromFile: string, spec: string, ts: TSConfig): string | null {
  // very light: handle relative + baseUrl + simple paths['alias/*'] => 'src/*'
  if (spec.startsWith("."))
    return resolve(dirname(fromFile), spec + (spec.endsWith(".ts") ? "" : ""));
  const base = ts.baseUrl ? resolve(root, ts.baseUrl) : root;
  // paths alias
  const paths = ts.paths ?? {};
  for (const [alias, targets] of Object.entries(paths)) {
    const re = new RegExp("^" + alias.replace("*", "(.*)") + "$");
    const m = spec.match(re);
    if (m) {
      const tail = m[1] ?? "";
      const first = targets[0].replace("*", tail);
      return resolve(base, first);
    }
  }
  // bare import -> try baseUrl
  return resolve(base, spec);
}

export function resolvePyImport(root: string, fromFile: string, stmtText: string): string | null {
  // crude: handle `from .foo import bar` and `import pkg.mod`
  const rel = stmtText.match(/from\s+(\.+[A-Za-z0-9_\.]*)\s+import/)?.[1];
  if (rel?.startsWith(".")) {
    let up = rel.match(/^(\.+)/)?.[1].length ?? 1;
    let dir = dirname(fromFile);
    while (up-- > 0) dir = dirname(dir);
    const mod = rel.replace(/^\.+/, "").replace(/\./g, "/");
    return resolve(dir, mod + ".py");
  }
  const absMod = stmtText.match(/import\s+([A-Za-z0-9_\.]+)/)?.[1];
  if (absMod) return resolve(root, absMod.replace(/\./g, "/") + ".py");
  return null;
}

export function readGoModule(root: string): string | null {
  try {
    const text = readFileSync(resolve(root, "go.mod"), "utf8");
    const m = text.match(/^module\s+(.+)$/m);
    return m ? m[1].trim() : null;
  } catch { return null; }
}

export function resolveGoImport(root: string, fromFile: string, spec: string, modulePath: string | null): string | null {
  if (spec.startsWith(".")) return resolve(dirname(fromFile), spec);
  if (modulePath && spec.startsWith(modulePath + "/")) {
    const sub = spec.slice(modulePath.length + 1);
    return resolve(root, sub);
  }
  // try vendor or local matching folder
  return resolve(root, spec);
}
```

### `tests/incremental-index.test.ts`

```ts
import { describe, it, expect } from "vitest";
import { buildIndex, updateIndex, getIndex } from "../packages/repo-index/src/index";
import { resolve } from "path";
import { writeFileSync, readFileSync } from "fs";

describe("incremental indexing", async () => {
  it("re-parses only changed files", async () => {
    const root = resolve("fixtures/ts-mini");
    await buildIndex(root);
    const before = getIndex();
    const target = resolve(root, "src/mathy.ts");
    const orig = readFileSync(target, "utf8");
    writeFileSync(target, orig + "\n// touch");
    const after = await updateIndex([target]);
    expect(after.updatedAt).toBeGreaterThan(before.updatedAt);
    // sanity: symbol count should be similar
    expect(after.symbols.length).toBeGreaterThan(0);
    // restore
    writeFileSync(target, orig);
  });
});
```
