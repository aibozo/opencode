## Objectives

* **Record** what code the model actually saw (initial FULL ranges + any `readSpan` fetches).
* **Measure** whether every edited line came from that exposure set (**line-recall = 100%**).
* **Auto-expand** the frontier when recall < 100% (bump neighbor radius or nudge spans).
* Provide a **mini eval harness** to simulate sessions on fixtures using expected patches.

### `packages/metrics/src/exposure.ts`

```ts
import { readFileSync, writeFileSync, mkdirSync, existsSync } from "fs";
import { dirname } from "path";
import type { ExposureLog, ExposureRange } from "./types";

const EXP_PATH = ".opencode/state/exposure.jsonl";

export function appendExposure(log: ExposureLog) {
  mkdirSync(dirname(EXP_PATH), { recursive: true });
  writeFileSync(EXP_PATH, JSON.stringify(log) + "\n", { flag: "a" });
}

export function readExposures(sessionId: string): ExposureLog[] {
  if (!existsSync(EXP_PATH)) return [];
  const lines = readFileSync(EXP_PATH, "utf8").trim().split("\n").filter(Boolean);
  return lines.map(l => JSON.parse(l) as ExposureLog).filter(x => x.sessionId === sessionId);
}

export function mergeExposures(logs: ExposureLog[]): ExposureRange[] {
  // union all ranges per path
  const map = new Map<string, [number, number][]>();
  const push = (p: string, start: number, end: number) => {
    if (!map.has(p)) map.set(p, []);
    map.get(p)!.push([start, end]);
  };
  for (const l of logs) {
    l.full.forEach(r => push(r.path, r.start, r.end));
    l.spans.forEach(r => push(r.path, r.start, r.end));
  }
  // merge overlaps
  const out: ExposureRange[] = [];
  for (const [path, ranges] of map) {
    const sorted = ranges.sort((a,b)=>a[0]-b[0]);
    let [s,e] = sorted[0];
    for (let i=1;i<sorted.length;i++){
      const [cs,ce] = sorted[i];
      if (cs <= e+1) e = Math.max(e, ce);
      else { out.push({ path, start: s, end: e }); [s,e] = [cs,ce]; }
    }
    out.push({ path, start: s, end: e });
  }
  return out;
}
```

### `packages/metrics/src/git.ts`

```ts
import { execSync } from "child_process";
import type { EditHunk } from "./types";

export function diffHunks(baseRef = "HEAD~1", headRef = "HEAD"): EditHunk[] {
  const out = execSync(`git diff --unified=0 ${baseRef} ${headRef}`, { encoding: "utf8" });
  // parse @@ -a,b +c,d @@  file headers come earlier with ---/+++ lines
  const hunks: EditHunk[] = [];
  let currentPath: string | null = null;
  const lines = out.split("\n");
  for (const line of lines) {
    if (line.startsWith("+++ b/")) { currentPath = line.slice(6).trim(); continue; }
    const m = line.match(/^@@ [^+]+\+(\d+)(?:,(\d+))? @@/);
    if (m && currentPath) {
      const start = parseInt(m[1], 10);
      const len = m[2] ? parseInt(m[2], 10) : 1;
      hunks.push({ path: currentPath, start, end: start + Math.max(0, len - 1) });
    }
  }
  return hunks;
}
```

### `packages/metrics/src/recall.ts`

```ts
import type { EditHunk, ExposureRange } from "./types";

export interface RecallResult { total: number; covered: number; pct: number; misses: EditHunk[]; }

export function computeRecall(edits: EditHunk[], exposure: ExposureRange[]): RecallResult {
  const coverMap = new Map<string, ExposureRange[]>();
  exposure.forEach(r => {
    const arr = coverMap.get(r.path) ?? [];
    arr.push(r); coverMap.set(r.path, arr);
  });

  let covered = 0;
  const misses: EditHunk[] = [];
  for (const h of edits) {
    const ranges = (coverMap.get(h.path) ?? []).sort((a,b)=>a.start-b.start);
    let ok = false;
    for (const r of ranges) {
      if (h.start >= r.start && h.end <= r.end) { ok = true; break; }
    }
    if (ok) covered++;
    else misses.push(h);
  }
  const total = edits.length;
  const pct = total ? Math.round((covered/total)*1000)/10 : 100;
  return { total, covered, pct, misses };
}
```

### `packages/repo-mcp/src/logging.ts`

```ts
import { appendExposure } from "@opencode/metrics/src/exposure";
import type { ContextPack } from "@opencode/shared/src/types";

export function logWorkingSet(sessionId: string, root: string, pack: ContextPack, fileLengths: Map<string, number>) {
  const full = pack.full.flatMap(f => {
    if (!f.ranges?.length) {
      // whole file exposure, use 1..N (from fileLengths map; fallback to large number)
      const n = fileLengths.get(f.path) ?? 1_000_000;
      return [{ path: f.path, start: 1, end: n }];
    }
    return f.ranges.map(([a,b]) => ({ path: f.path, start: a, end: b }));
  });
  appendExposure({ sessionId, root, full, spans: [], when: new Date().toISOString() });
}

export function logSpan(sessionId: string, root: string, path: string, start: number, end: number) {
  appendExposure({ sessionId, root, full: [], spans: [{ path, start, end }], when: new Date().toISOString() });
}
```

```ts
case "repo.session.start": {
  const _ = ZSessionStartArgs.parse(args ?? {});
  const sessionId = uuidv4();
  return { sessionId };
}
case "repo.workingSet.select": {
  const p = ZSelectArgs.parse(args ?? {});
  const { pack, debug } = await selectWorkingSet(root, p);
  // estimate line counts per file for whole-file exposure
  const fileLens = new Map<string, number>();
  for (const f of pack.full) {
    const rel = f.path;
    const abs = resolve(root, rel);
    const nLines = readFileSync(abs, "utf8").split(/\r?\n/).length;
    fileLens.set(rel, nLines);
  }
  const sessionId = (args?.sessionId as string) || (process.env.SESSION_ID ?? "");
  if (sessionId) logWorkingSet(sessionId, root, pack, fileLens);
  return { pack, debug };
}
case "repo.code.readSpan": {
  const p = ZReadSpanArgs.parse(args ?? {});
  const { abs, rel } = normalizeAndCheck(root, p.path);
  const result = readSpan(abs, p.start, p.end);
  const sha1 = crypto.createHash("sha1").update(result.text).digest("hex");
  const sessionId = (args?.sessionId as string) || (process.env.SESSION_ID ?? "");
  if (sessionId) logSpan(sessionId, root, rel, p.start, p.end);
  return { path: rel, start: p.start, end: p.end, lines: result.total, sha1, text: result.text };
}
case "repo.metrics.checkRecall": {
  const p = ZRecallArgs.parse(args ?? {});
  const logs = readExposures(p.sessionId);
  const exposure = mergeExposures(logs);
  const edits = diffHunks(p.baseRef, p.headRef);
  const recall = computeRecall(edits, exposure);
  return { recall };
}
case "repo.frontier.expand": {
  const p = ZExpandArgs.parse(args ?? {});
  // grab last selection’s radius (we don’t store it yet), so just bump by delta
  const { pack, debug } = await selectWorkingSet(root, { task: p.task, radius: 2, budgetTokens: p.budgetTokens ?? 64000, preferSpanContext: true });
  // logging like above
  const fileLens = new Map<string, number>();
  for (const f of pack.full) {
    const n = readFileSync(resolve(root, f.path), "utf8").split(/\r?\n/).length;
    fileLens.set(f.path, n);
  }
  logWorkingSet(p.sessionId, root, pack, fileLens);
  return { pack, debug };
}
```

**Note:** We keep session id optional; agent should call `repo.session.start` once and pass `sessionId` with `workingSet.select` / `readSpan`.

```ts
import { execSync } from "child_process";
import { readFileSync } from "fs";

export function applyPatch(diffPath: string) {
  const diff = readFileSync(diffPath, "utf8");
  execSync("git diff --quiet || true"); // ignore dirty state in CI
  execSync("git add -A && git commit -m 'baseline' || true");
  execSync("git apply --reject --unidiff-zero -p0", { input: diff, stdio: ["pipe","inherit","inherit"] });
  execSync("git add -A && git commit -m 'apply expected patch'");
}
```

### `packages/eval-runner/src/run.ts`

```ts
import { spawn } from "child_process";
import { resolve } from "path";
import { readFileSync } from "fs";
import { applyPatch } from "./patch";

async function rpc(proc: any, id: number, method: string, params?: any) {
  proc.stdin.write(JSON.stringify({ jsonrpc: "2.0", id, method, params }) + "\n");
  const line: string = await new Promise(res => proc.stdout.once("data", (b: Buffer) => res(b.toString("utf8").trim())));
  return JSON.parse(line);
}

export async function runFixtureEval(fixDir: string, issueFile = "ISSUE.md", diffFile = "expected_patch.diff") {
  const proc = spawn("node", ["packages/repo-mcp/dist/bin/repo-mcp.js"], { stdio: ["pipe","pipe","pipe"], env: { ...process.env, REPO_ROOT: resolve(fixDir) } });
  await rpc(proc, 1, "tools/call", { name: "repo.graphBuild", arguments: {} });
  const ses = await rpc(proc, 2, "tools/call", { name: "repo.session.start", arguments: {} });
  const sessionId = ses.result.sessionId;

  const issue = readFileSync(resolve(fixDir, issueFile), "utf8");
  await rpc(proc, 3, "tools/call", { name: "repo.workingSet.select", arguments: { task: issue, budgetTokens: 4000, radius: 1, sessionId } });

  // Simulate that the agent later read a small span (typical behavior). Optional, skip if not needed.

  // Apply the expected patch and compute recall
  applyPatch(resolve(fixDir, diffFile));
  const r = await rpc(proc, 4, "tools/call", { name: "repo.metrics.checkRecall", arguments: { sessionId } });
  proc.kill();
  return r.result.recall;
}
```

```ts
import { describe, it, expect } from "vitest";
import { mergeExposures, readExposures } from "../packages/metrics/src/exposure";
import { computeRecall } from "../packages/metrics/src/recall";
import { diffHunks } from "../packages/metrics/src/git";

describe("recall computation", () => {
  it("computes 100% recall when all edits are exposed", () => {
    const exposure = [{ path: "src/x.ts", start: 10, end: 50 }];
    const edits = [{ path: "src/x.ts", start: 12, end: 12 }];
    const r = computeRecall(edits, exposure);
    expect(r.pct).toBe(100);
  });
  it("detects misses", () => {
    const exposure = [{ path: "src/x.ts", start: 10, end: 20 }];
    const edits = [{ path: "src/x.ts", start: 25, end: 26 }];
    const r = computeRecall(edits, exposure);
    expect(r.pct).toBeLessThan(100);
    expect(r.misses.length).toBe(1);
  });
});
```

## PR description (paste)

**Title:** PR 7 — Safety net & metrics: line-recall, exposure logging, frontier auto-expand, eval harness

**Summary:**
Adds a metrics layer to guarantee we don’t silently drop necessary context. The MCP server now logs exposures (initial FULL ranges + subsequent `readSpan`s) by session, and can compute **line-recall** by diffing the applied patch. On misses, a `frontier.expand` tool returns a broader working set. A small eval runner applies fixture patches and verifies recall. Tests cover recall math, expansion, and end-to-end eval.

**How to test:**

1. `bun run build`
2. `bun run test`
3. (Optional) `node packages/repo-mcp/dist/bin/repo-mcp.js` and drive the new tools manually.
