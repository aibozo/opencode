
# PR 6 — Extractive NL Compression + Real Token Counting

## Objectives

* Add a **faithful, extractive** compressor for issues/logs/traces: shrink 2–5× with **zero paraphrase**.
* Provide a **real tokenizer** (tiktoken) with a safe **heuristic fallback**.
* Use the compressor inside **`selectWorkingSet`** to produce tighter `pack.notes`.
* (Optional) Expose `repo.nl.compress` as an MCP tool for ad-hoc use.

---

* Upgrade `tokenize.ts` to support **tiktoken** when present.
* Update `working-set.ts` to call `compressNotes(...)` instead of the stub.
* Extend MCP with a new tool: `repo.nl.compress(text, budgetTokens)` (optional but nice).
* Tests: faithfulness, determinism, budget adherence, edge cases.

```ts
export interface Tokenizer {
  estimateTokens(text: string): number;
}

let _bpe: any | null = null;

export async function createTokenizer(): Promise<Tokenizer> {
  if (_bpe) return _bpe;

  try {
    // Lazy init tiktoken (optional)
    const { encoding_for_model } = await import("@dqbd/tiktoken");
    const enc = encoding_for_model("gpt-4o"); // close enough; model here is OpenAI-first
    _bpe = {
      estimateTokens(text: string) {
        try { return enc.encode(text).length; }
        catch { return heuristicEstimate(text); }
      }
    };
    return _bpe;
  } catch {
    // Fallback heuristic
    _bpe = { estimateTokens: heuristicEstimate };
    return _bpe;
  }
}

function heuristicEstimate(text: string): number {
  const punct = (text.match(/[{}()[\].,;:+\-*/=<>]/g) ?? []).length;
  const base = Math.ceil(text.length / 4);
  return base + Math.ceil(punct / 3);
}
```

> Note: We keep the exported `Tokenizer` type and add `createTokenizer()` so both `working-set` and `nl-compress` use the same counter.

````ts
// Sentence-ish segmentation that preserves code/stack blocks as atomic units.
export interface Segment { text: string; start: number; end: number; kind: "line"|"code"|"stack"|"para"; }

export function segment(text: string): Segment[] {
  const lines = text.split(/\r?\n/);

  // Mark fenced code blocks
  const segs: Segment[] = [];
  let i = 0;
  while (i < lines.length) {
    // fenced code ```...```
    if (/^\s*```/.test(lines[i])) {
      const start = i;
      i++;
      while (i < lines.length && !/^\s*```/.test(lines[i])) i++;
      const end = Math.min(i, lines.length - 1);
      segs.push({ text: lines.slice(start, end + 1).join("\n"), start, end, kind: "code" });
      i++; continue;
    }
    // stack trace-ish lines cluster
    if (/(^Traceback|^\s*at\s|File ".*", line \d+)/.test(lines[i])) {
      const start = i;
      let end = i;
      while (end + 1 < lines.length && /(^\s+at\s|^\s*File ".*", line \d+|^\s+.*Error|^\s*Traceback|^\s*Caused by:)/.test(lines[end + 1])) end++;
      segs.push({ text: lines.slice(start, end + 1).join("\n"), start, end, kind: "stack" });
      i = end + 1; continue;
    }
    // paragraph chunk
    const start = i;
    while (i + 1 < lines.length && lines[i + 1].trim() !== "") i++;
    const end = i;
    const block = lines.slice(start, end + 1).join("\n").trim();
    if (block) segs.push({ text: block, start, end, kind: "para" });
    i++;
    // skip blank
    while (i < lines.length && lines[i].trim() === "") i++;
  }
  return segs;
}
````

**`packages/nl-compress/src/score.ts`**

```ts
import type { Segment } from "./segment";

export interface ScoreOpts { keywords?: string[]; }

export function scoreSegments(segs: Segment[], opts: ScoreOpts = {}): number[] {
  const scores: number[] = [];
  const kws = new Set((opts.keywords ?? []).map(k => k.toLowerCase()));

  const total = segs.length || 1;
  segs.forEach((s, idx) => {
    let sc = 0;

    // type prior
    if (s.kind === "stack") sc += 6;
    if (s.kind === "code") sc += 3;
    if (s.kind === "para") sc += 2;

    // position (earlier gets small boost)
    sc += (total - idx) / total; // 0..1

    // keyword hits
    const lower = s.text.toLowerCase();
    let hits = 0;
    kws.forEach(k => { if (lower.includes(k)) hits++; });
    sc += Math.min(4, hits); // cap

    // error words
    if (/(error|exception|assert|fail|expected|received)/i.test(s.text)) sc += 2;

    scores.push(sc);
  });
  return scores;
}
```

**`packages/nl-compress/src/mmr.ts`**

```ts
// Maximal Marginal Relevance over bag-of-words Jaccard similarity
import type { Segment } from "./segment";

function tokenSet(s: string): Set<string> {
  return new Set((s.toLowerCase().match(/[a-z0-9_]+/g) ?? []).filter(w => w.length > 2));
}
function jaccard(a: Set<string>, b: Set<string>) {
  const inter = new Set([...a].filter(x => b.has(x)));
  const uni = new Set([...a, ...b]);
  return inter.size / Math.max(1, uni.size);
}

export function selectMMR(segs: Segment[], scores: number[], lambda = 0.8, limit?: number): number[] {
  const idxs = segs.map((_, i) => i).sort((a, b) => scores[b] - scores[a]);
  const chosen: number[] = [];
  const sets = segs.map(s => tokenSet(s.text));
  while (idxs.length) {
    let best = idxs[0], bestVal = -Infinity;
    for (const i of idxs) {
      const sim = chosen.length ? Math.max(...chosen.map(j => jaccard(sets[i], sets[j]))) : 0;
      const val = lambda * scores[i] - (1 - lambda) * sim;
      if (val > bestVal) { bestVal = val; best = i; }
    }
    chosen.push(best);
    idxs.splice(idxs.indexOf(best), 1);
    if (limit && chosen.length >= limit) break;
  }
  // preserve original order when emitting
  return chosen.sort((a, b) => segs[a].start - segs[b].start);
}
```

**`packages/nl-compress/src/index.ts`**

```ts
import { segment } from "./segment";
import { scoreSegments } from "./score";
import { selectMMR } from "./mmr";
import type { Segment } from "./segment";
import { createTokenizer } from "@opencode/repo-index/src/tokenize";

export interface CompressOpts {
  budgetTokens: number;
  keywords?: string[];
  keepAtLeast?: number;      // default 3 segments
  lambda?: number;           // MMR tradeoff (0..1), default 0.8
}

export async function compress(text: string, opts: CompressOpts): Promise<string> {
  const tok = await createTokenizer();
  const segs = segment(text);
  if (!segs.length) return "";

  // Score + select
  const scores = scoreSegments(segs, { keywords: opts.keywords });
  const order = selectMMR(segs, scores, opts.lambda ?? 0.8);

  // Pack until budget, preserving original order; always include at least N segments
  const keepMin = Math.max(1, opts.keepAtLeast ?? 3);
  const out: string[] = [];
  let used = 0; let kept = 0;
  for (const i of order) {
    const s = segs[i];
    const add = s.text;
    const cost = tok.estimateTokens(add);
    if (kept < keepMin || used + cost <= opts.budgetTokens) {
      out.push(add);
      used += cost; kept++;
    } else {
      break;
    }
  }

  // Faithful extractive: join with blank lines, no paraphrase
  return out.join("\n\n");
}
```

  });
