### `packages/repo-mcp/src/server.ts`

```ts
// Minimal MCP-ish stdio loop (PR 0). Later PRs will speak real MCP JSON-RPC 2.0.
import { handleGraphBuild } from "./tools";

// A trivial, line-delimited protocol for smoke tests (not real MCP yet).
// Request: {"method":"repo.graphBuild","params":{"root":"..."}}
export async function startServer() {
  const rl = require("readline").createInterface({ input: process.stdin, output: process.stdout });
  console.log("MCP:READY"); // test looks for this
  rl.on("line", async (line: string) => {
    try {
      const msg = JSON.parse(line);
      if (msg?.method === "repo.graphBuild") {
        const res = await handleGraphBuild(msg.params ?? {});
        process.stdout.write(JSON.stringify({ id: msg.id ?? null, result: res }) + "\n");
      } else {
        process.stdout.write(JSON.stringify({ id: msg.id ?? null, error: "unknown_method" }) + "\n");
      }
    } catch (e) {
      process.stdout.write(JSON.stringify({ error: "bad_json", detail: String(e) }) + "\n");
    }
  });
}
```

### `tests/prefix-hash.test.ts`

> Ensures our *stable* instruction prefix doesn’t accidentally change between runs.

```ts
import { describe, it, expect } from "vitest";
import { createHash } from "crypto";
import { readFileSync } from "fs";

function stableHash() {
  const a = readFileSync("AGENTS.md", "utf-8");
  const b = readFileSync("docs/dev-standards.md", "utf-8");
  return createHash("sha256").update(a + "\n---\n" + b).digest("hex");
}

describe("stable instruction prefix", () => {
  it("produces a deterministic hash", () => {
    const h1 = stableHash();
    const h2 = stableHash();
    expect(h1).toBe(h2);
  });
});
```

### `tests/mcp-smoke.test.ts`

> Launches the MCP stub and checks for readiness + a basic request/response.

```ts
import { describe, it, expect } from "vitest";
import { spawn } from "child_process";

function send(proc: any, obj: any) {
  proc.stdin.write(JSON.stringify(obj) + "\n");
}

function recv(proc: any): Promise<any> {
  return new Promise((resolve) => {
    proc.stdout.once("data", (buf: Buffer) => resolve(buf.toString().trim()));
  });
}

describe("repo-mcp smoke", () => {
  it("starts and handles repo.graphBuild", async () => {
    const proc = spawn("bun", ["run", "dev:repo-mcp"], { stdio: ["pipe", "pipe", "pipe"] });

    const ready = await recv(proc);
    expect(ready).toContain("MCP:READY");

    send(proc, { id: 1, method: "repo.graphBuild", params: { root: process.cwd() } });
    const resLine = await recv(proc);
    const res = JSON.parse(resLine);
    expect(res.id).toBe(1);
    expect(res.result?.ok).toBe(true);

    proc.kill();
  });
});
```

## How to run locally (for your agent / CI)

1. **Install**

   * Install Bun ≥ 1.1 and Node ≥ 20.
   * `bun install`

2. **Build**

   * `bun run build`

3. **Run tests**

   * `bun run test`

4. **Smoke the MCP stub manually (optional)**

   * `bun run dev:repo-mcp`
   * In another shell:
     `echo '{"id":1,"method":"repo.graphBuild","params":{"root":"."}}' | nc localhost 0` *(or just type JSON then Enter if you’re piping stdin manually)*
